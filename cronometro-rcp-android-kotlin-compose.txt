    inner class LocalBinder : Binder() {
        fun getService(): RcpForegroundService = this@RcpForegroundService
    }

    private val binder = LocalBinder()
    private val serviceScope = CoroutineScope(SupervisorJob() + Dispatchers.Main.immediate)

    private lateinit var repository: RcpRepository
    private lateinit var audioEngine: RcpAudioEngine
    private lateinit var mediaSession: MediaSessionCompat
    private var wakeLock: PowerManager.WakeLock? = null

    private val _timerState = MutableStateFlow(RcpTimerState())
    val timerState: StateFlow<RcpTimerState> = _timerState.asStateFlow()

    private var timerJob: Job? = null
    private var metronomeJob: Job? = null
    private var mp3SyncJob: Job? = null

    private var sessionStartedAtIso: String = ""
    private var lastAnnouncementCycleRemaining: Int = -1
    private var thirtySecondAnnouncedCycle: Int = -1
    private var lastReminderElapsed = mutableMapOf<ReminderKey, Int>()
    private var beatStepInPattern = 0
    private var isForegroundActive = false

    private var amiodaronaDose1Prompted = false
    private var amiodaronaDose2Prompted = false
    private var fiveHsInitialDone = false

    private fun estimateSpeechMs(text: String): Long {
        val len = text.trim().length
        return (len * 65L + 400L).coerceIn(1600L, 7000L)
    }

    private fun speakProtected(text: String, flushQueue: Boolean = false) {
        if (text.isBlank()) return
        val nowMs = SystemClock.elapsedRealtime()
        val est = estimateSpeechMs(text)
        protectedSpeechUntilMs = maxOf(if (flushQueue) nowMs else protectedSpeechUntilMs, nowMs + est)
        audioEngine.speak(text, flushQueue = flushQueue)
    }

    override fun onCreate() {
        super.onCreate()
        repository = RcpRepository(this)
        audioEngine = RcpAudioEngine(this)
        audioEngine.updateAudioSettings(repository.loadAudioSettings())

        createNotificationChannel()
        mediaSession = MediaSessionCompat(this, "CronometroRcpMediaSession").apply {
            setCallback(object : MediaSessionCompat.Callback() {
                override fun onPlay() = resumeRcp()
                override fun onPause() = pauseRcp()
                override fun onStop() = stopAndSaveSession()
            })
            isActive = true
        }

        val initialMode = repository.loadSelectedMode()
        val cfg = repository.loadModeConfigs()[initialMode]!!
        _timerState.value = RcpTimerState(
            activeMode = initialMode,
            cycleRemainingSeconds = cfg.cycleDurationSeconds,
            adrenalinaRemainingSeconds = cfg.reminders[ReminderKey.ADRENALINA]?.intervalSeconds
        )
    }

    override fun onBind(intent: Intent?): IBinder = binder

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        when (intent?.action) {
            ACTION_PAUSE -> pauseRcp()
            ACTION_RESUME -> resumeRcp()
            ACTION_STOP -> stopAndSaveSession()
        }
        return START_STICKY
    }

    fun selectMode(mode: RcpMode) {
        if (_timerState.value.isRunning) return
        repository.saveSelectedMode(mode)
        val cfg = repository.loadModeConfigs()[mode]!!
        _timerState.value = _timerState.value.copy(
            activeMode = mode,
            cycleRemainingSeconds = cfg.cycleDurationSeconds,
            adrenalinaRemainingSeconds = cfg.reminders[ReminderKey.ADRENALINA]?.intervalSeconds,
            currentPhaseLabel = "Pronto"
        )
    }

    private fun buildSyncSignature(s: AudioSettings): String =
        "${s.rhythmSource.name}:${s.mp3UriString}:${s.mp3ExactBpm}:${s.mp3OriginalBpm}:${s.mp3TrimStartMs}:${s.mp3LoopEndMs}:${s.mp3SyncSpeed}:${s.followMusicRhythm}"

    fun refreshAudioSettings(settings: AudioSettings) {
        liveBeatOffsetMs = settings.mp3BeatOffsetMs
        val nextSig = buildSyncSignature(settings)
        audioEngine.updateAudioSettings(settings)
        val state = _timerState.value
        if (state.isRunning && !state.isPaused && nextSig != lastSyncSignature) {
            lastSyncSignature = nextSig
            val cfg = repository.loadModeConfigs()[state.activeMode]!!
            pausedMp3PosMs = if (audioEngine.isMp3PlayingNow()) audioEngine.getMp3PositionMs() else null
            launchLoops(cfg)
        }
    }

    fun startRcp() {
        val mode = _timerState.value.activeMode
        val cfg = repository.loadModeConfigs()[mode]!!
        sessionStartedAtIso = Instant.now().toString()
        lastAnnouncementCycleRemaining = -1
        thirtySecondAnnouncedCycle = -1
        lastReminderElapsed.clear()
        beatStepInPattern = 0

        amiodaronaDose1Prompted = false
        amiodaronaDose2Prompted = false
        fiveHsInitialDone = false
        fiveHsTipIndex = 0
        lastFiveHsTipAt = 0
        pendingFiveHsTip = false
        lastCriticalSpeechAt = 0
        isVentilationPhase = false
        lastCompressorPreWarnAnchor = -1
        lastAdrenalinaPreWarnAnchor = -1
        ventilationWaitUntilMs = 0L
        remainingVentPauseOnPauseMs = 0L
        protectedSpeechUntilMs = 0L
        pausedMp3PosMs = null
        val startAudioSettings = repository.loadAudioSettings()
        liveBeatOffsetMs = startAudioSettings.mp3BeatOffsetMs
        lastSyncSignature = buildSyncSignature(startAudioSettings)

        _timerState.value = RcpTimerState(
            isRunning = true,
            isPaused = false,
            activeMode = mode,
            cycleNumber = 1,
            cycleRemainingSeconds = cfg.cycleDurationSeconds,
            totalElapsedSeconds = 0,
            currentPhaseLabel = if (cfg.pattern.isContinuous) "Compressões contínuas" else "Compressões",
            adrenalinaRemainingSeconds = cfg.reminders[ReminderKey.ADRENALINA]?.let {
                if (it.enabled) it.intervalSeconds else null
            },
            adrenalinaDue = false,
            isIntubated = false,
            shocksCount = 0,
            amiodaronaCount = 0,
            currentFiveHsTsTip = null,
            events = emptyList()
        )

        promoteToForeground()
        acquireWakeLock()
        updateMediaPlaybackState(isPlaying = true)

        speakProtected("RCP iniciada. ${cfg.voiceMessages.compressionsCue}", flushQueue = true)
        launchLoops(cfg)
    }

    fun pauseRcp() {
        val current = _timerState.value
        if (!current.isRunning || current.isPaused) return
        if (ventilationWaitUntilMs > 0L) {
            remainingVentPauseOnPauseMs = (ventilationWaitUntilMs - SystemClock.elapsedRealtime()).coerceAtLeast(0L)
            ventilationWaitUntilMs = 0L
        }
        pausedMp3PosMs = if (audioEngine.isMp3PlayingNow()) audioEngine.getMp3PositionMs() else pausedMp3PosMs
        timerJob?.cancel()
        metronomeJob?.cancel()
        mp3SyncJob?.cancel()
        audioEngine.pauseMp3Loop()
        _timerState.value = current.copy(isPaused = true)
        updateMediaPlaybackState(isPlaying = false)
        updateNotification()
        audioEngine.speak("Cronômetro pausado.", flushQueue = true)
    }

    fun resumeRcp() {
        val current = _timerState.value
        if (!current.isRunning || !current.isPaused) return
        if (remainingVentPauseOnPauseMs > 0L) {
            ventilationWaitUntilMs = SystemClock.elapsedRealtime() + remainingVentPauseOnPauseMs
            remainingVentPauseOnPauseMs = 0L
        }
        val cfg = repository.loadModeConfigs()[current.activeMode]!!
        _timerState.value = current.copy(isPaused = false)
        updateMediaPlaybackState(isPlaying = true)
        updateNotification()
        audioEngine.speak("Retomando RCP.", flushQueue = true)
        launchLoops(cfg)
    }

    fun registerEvent(type: RcpEventType, customTitle: String? = null) {
        val current = _timerState.value
        val cfg = repository.loadModeConfigs()[current.activeMode]!!
        val record = RcpEventRecord(
            type = type,
            customTitle = customTitle,
            elapsedSeconds = current.totalElapsedSeconds,
            cycleNumber = current.cycleNumber,
            timestampIso = Instant.now().toString()
        )

        val nextShocks = if (type == RcpEventType.CHOQUE) current.shocksCount + 1 else current.shocksCount
        val nextAmio = if (type == RcpEventType.AMIODARONA) current.amiodaronaCount + 1 else current.amiodaronaCount
        val nextAdreDue = if (type == RcpEventType.ADRENALINA) false else current.adrenalinaDue
        val nextIntubated = if (type == RcpEventType.INTUBACAO) true else current.isIntubated

        if (type == RcpEventType.INTUBACAO) {
            ventilationWaitUntilMs = 0L
            remainingVentPauseOnPauseMs = 0L
            isVentilationPhase = false
            beatStepInPattern = 0
        }

        _timerState.value = current.copy(
            shocksCount = nextShocks,
            amiodaronaCount = nextAmio,
            adrenalinaDue = nextAdreDue,
            isIntubated = nextIntubated,
            currentPhaseLabel = if (type == RcpEventType.INTUBACAO) "Compressões contínuas (Intubado)" else current.currentPhaseLabel,
            events = listOf(record) + current.events
        )

        val spoken = if (type == RcpEventType.PERSONALIZADO && !customTitle.isNullOrBlank()) {
            "$customTitle registrado."
        } else {
            type.spokenConfirmation
        }
        lastCriticalSpeechAt = current.totalElapsedSeconds
        speakProtected(spoken, flushQueue = false)

        if (type == RcpEventType.CHOQUE && cfg.amiodaronaProtocol.enabled) {
            if (nextShocks >= cfg.amiodaronaProtocol.firstDoseAfterShock && nextAmio == 0 && !amiodaronaDose1Prompted) {
                amiodaronaDose1Prompted = true
                lastCriticalSpeechAt = current.totalElapsedSeconds
                speakProtected(cfg.amiodaronaProtocol.firstDoseMessage, flushQueue = false)
            } else if (nextShocks >= cfg.amiodaronaProtocol.secondDoseAfterShock && nextAmio == 1 && !amiodaronaDose2Prompted) {
                amiodaronaDose2Prompted = true
                lastCriticalSpeechAt = current.totalElapsedSeconds
                speakProtected(cfg.amiodaronaProtocol.secondDoseMessage, flushQueue = false)
            }
        }

        when (type) {
            RcpEventType.ADRENALINA -> lastReminderElapsed[ReminderKey.ADRENALINA] = current.totalElapsedSeconds
            RcpEventType.AMIODARONA -> lastReminderElapsed[ReminderKey.AMIODARONA] = current.totalElapsedSeconds
            RcpEventType.OUTRO_ANTIARRITMICO -> lastReminderElapsed[ReminderKey.ANTIARRITMICO] = current.totalElapsedSeconds
            RcpEventType.CHOQUE -> lastReminderElapsed[ReminderKey.CHOQUE] = current.totalElapsedSeconds
            RcpEventType.VERIFICAR_RITMO -> lastReminderElapsed[ReminderKey.VERIFICAR_RITMO] = current.totalElapsedSeconds
            RcpEventType.TROCAR_COMPRESSOR -> lastReminderElapsed[ReminderKey.TROCAR_COMPRESSOR] = current.totalElapsedSeconds
            else -> Unit
        }
    }

    fun stopAndSaveSession(): RcpSessionRecord? {
        ventilationWaitUntilMs = 0L
        remainingVentPauseOnPauseMs = 0L
        pausedMp3PosMs = null
        beatStepInPattern = 0
        timerJob?.cancel()
        metronomeJob?.cancel()
        mp3SyncJob?.cancel()
        audioEngine.stopMp3Loop()
        val current = _timerState.value
        var saved: RcpSessionRecord? = null

        if (current.isRunning && current.totalElapsedSeconds > 0) {
            val shocks = current.events.count { it.type == RcpEventType.CHOQUE }
            val meds = current.events.count {
                it.type == RcpEventType.ADRENALINA ||
                    it.type == RcpEventType.AMIODARONA ||
                    it.type == RcpEventType.OUTRO_ANTIARRITMICO
            }
            val switches = current.events.count { it.type == RcpEventType.TROCAR_COMPRESSOR }

            saved = RcpSessionRecord(
                startedAtIso = sessionStartedAtIso.ifBlank { Instant.now().toString() },
                mode = current.activeMode,
                totalDurationSeconds = current.totalElapsedSeconds,
                completedCycles = (current.cycleNumber - 1).coerceAtLeast(0),
                shocksCount = shocks,
                medicationsCount = meds,
                compressorSwitchesCount = switches,
                events = current.events.reversed()
            )
            repository.appendSession(saved)
        }

        val cfg = repository.loadModeConfigs()[current.activeMode]!!
        _timerState.value = RcpTimerState(
            isRunning = false,
            isPaused = false,
            activeMode = current.activeMode,
            cycleNumber = 1,
            cycleRemainingSeconds = cfg.cycleDurationSeconds,
            totalElapsedSeconds = 0,
            currentPhaseLabel = "Pronto",
            adrenalinaRemainingSeconds = cfg.reminders[ReminderKey.ADRENALINA]?.intervalSeconds,
            isIntubated = false,
            events = emptyList()
        )

        releaseWakeLock()
        updateMediaPlaybackState(isPlaying = false)
        if (isForegroundActive) {
            stopForeground(STOP_FOREGROUND_REMOVE)
            isForegroundActive = false
        }
        return saved
    }

    private fun launchLoops(cfg: ModeConfig) {
        timerJob?.cancel()
        metronomeJob?.cancel()
        mp3SyncJob?.cancel()

        timerJob = serviceScope.launch {
            var nextTickAt = SystemClock.elapsedRealtime() + 1000L
            while (isActive) {
                val waitMs = (nextTickAt - SystemClock.elapsedRealtime()).coerceAtLeast(0L)
                delay(waitMs)
                nextTickAt += 1000L
                onSecondTick(cfg)
            }
        }

        val settings = repository.loadAudioSettings()
        liveBeatOffsetMs = settings.mp3BeatOffsetMs
        val effectiveBpm = audioEngine.getEffectiveBeatBpm(cfg.metronomeBpm).coerceIn(60.0, 165.0)
        val beatPeriodRealMs = 60_000.0 / effectiveBpm
        val beatOriginMs = SystemClock.elapsedRealtime() + 35L

        metronomeJob = serviceScope.launch {
            var beatIdx = 0L
            while (isActive) {
                val targetAt = beatOriginMs + kotlin.math.round(beatIdx * beatPeriodRealMs).toLong()
                val waitMs = (targetAt - SystemClock.elapsedRealtime()).coerceAtLeast(0L)
                if (waitMs > 0L) delay(waitMs)
                onMetronomeBeat(cfg)
                beatIdx += 1L
            }
        }

        if (audioEngine.wantsMp3Playback()) {
            mp3SyncJob = serviceScope.launch {
                val exactMp3Bpm = if (settings.mp3SyncSpeed) {
                    audioEngine.getSyncedEffectiveMp3Bpm(cfg.metronomeBpm)
                } else {
                    audioEngine.getExactMp3Bpm()
                }
                val periodOrigMs = 60_000.0 / exactMp3Bpm
                val initialOffsetOrigMs = (((liveBeatOffsetMs.toDouble() % periodOrigMs) + periodOrigMs) % periodOrigMs)
                val trimStartMs = settings.mp3TrimStartMs.coerceAtLeast(0)
                val kFirstLoop = kotlin.math.max(
                    0,
                    kotlin.math.ceil((trimStartMs - initialOffsetOrigMs) / periodOrigMs - 1e-6).toInt()
                )
                val loopStartMs = kotlin.math.round(initialOffsetOrigMs + kFirstLoop * periodOrigMs).toInt()
                val loopEndMs = if (settings.mp3LoopEndMs > loopStartMs + (periodOrigMs * 4).toInt()) {
                    val nBeats = kotlin.math.max(
                        4,
                        kotlin.math.round((settings.mp3LoopEndMs - loopStartMs) / periodOrigMs).toInt()
                    )
                    kotlin.math.round(loopStartMs + nBeats * periodOrigMs).toInt()
                } else 0

                val startPosMs = pausedMp3PosMs ?: trimStartMs
                pausedMp3PosMs = null
                audioEngine.prepareMp3ForSyncedStart(cfg.metronomeBpm, startPosMs)

                val baseSpeed = audioEngine.getBasePlaybackSpeed(cfg.metronomeBpm).toDouble().coerceIn(0.85, 1.18)
                val distToNextBeatOrigMs = if (startPosMs <= loopStartMs) {
                    (loopStartMs - startPosMs).toDouble()
                } else {
                    val beatsPassed = (startPosMs - initialOffsetOrigMs) / periodOrigMs
                    (kotlin.math.ceil(beatsPassed - 1e-6) - beatsPassed) * periodOrigMs
                }
                val distRealMs = distToNextBeatOrigMs / baseSpeed
                val minTargetMs = SystemClock.elapsedRealtime() + 10.0 + distRealMs
                val kTargetBeat = kotlin.math.max(
                    1L,
                    kotlin.math.ceil((minTargetMs - beatOriginMs) / beatPeriodRealMs - 1e-6).toLong()
                )
                val targetBeatAt = beatOriginMs + kTargetBeat * beatPeriodRealMs
                val mp3StartAt = (targetBeatAt - distRealMs).toLong()
                val initialDelay = (mp3StartAt - SystemClock.elapsedRealtime()).coerceAtLeast(0L)
                if (initialDelay > 0L) delay(initialDelay)

                audioEngine.startMp3Now(cfg.metronomeBpm)
                val actualStartedElapsed = SystemClock.elapsedRealtime()
                var initialWarmupSeekDone = false
                var lastSeekAtMs = 0L
                var lastAppliedOffsetMs = liveBeatOffsetMs

                while (isActive) {
                    delay(45L)
                    if (!audioEngine.isMp3PlayingNow()) continue
                    val posMs = audioEngine.getMp3PositionMs()
                    if (loopEndMs > loopStartMs + 2000 && posMs >= loopEndMs) {
                        audioEngine.seekMp3To(loopStartMs)
                        continue
                    }

                    val nowMs = SystemClock.elapsedRealtime()
                    val currentOffsetMs = liveBeatOffsetMs
                    val offsetChangedByUser = currentOffsetMs != lastAppliedOffsetMs
                    val offsetOrigMs = (((currentOffsetMs.toDouble() % periodOrigMs) + periodOrigMs) % periodOrigMs)
                    val beatPhaseFrac = (nowMs - beatOriginMs) / beatPeriodRealMs
                    val targetFracInBeat = beatPhaseFrac - kotlin.math.floor(beatPhaseFrac)
                    val actualBeatsFromOffset = (posMs - offsetOrigMs) / periodOrigMs
                    val actualFracInBeat = actualBeatsFromOffset - kotlin.math.floor(actualBeatsFromOffset)

                    var fracError = targetFracInBeat - actualFracInBeat
                    if (fracError > 0.5) fracError -= 1.0
                    if (fracError < -0.5) fracError += 1.0
                    val errMs = fracError * periodOrigMs

                    if (offsetChangedByUser ||
                        (!initialWarmupSeekDone && nowMs - actualStartedElapsed in 250L..1600L && kotlin.math.abs(errMs) > 30.0) ||
                        (kotlin.math.abs(errMs) > 90.0 && nowMs - lastSeekAtMs > 300L)
                    ) {
                        lastAppliedOffsetMs = currentOffsetMs
                        initialWarmupSeekDone = true
                        lastSeekAtMs = nowMs
                        audioEngine.seekMp3To(kotlin.math.round(posMs + errMs).toInt().coerceAtLeast(0))
                        audioEngine.setPllSpeedTrim(1.0f)
                    } else if (errMs < -2.0) {
                        // MP3 adiantado: micro-redução suave (máx -3.5%)
                        val trim = (1.0 + errMs / 1800.0).coerceAtLeast(0.965)
                        audioEngine.setPllSpeedTrim(trim.toFloat())
                    } else if (errMs > 2.0) {
                        // MP3 atrasado: micro-aumento suave para alcançar a batida (máx +5.5%)
                        val trim = (1.0 + errMs / 900.0).coerceAtMost(1.055)
                        audioEngine.setPllSpeedTrim(trim.toFloat())
                    } else {
                        audioEngine.setPllSpeedTrim(1.0f)
                    }
                }
            }
        } else {
            audioEngine.pauseMp3Loop()
        }
    }

    private fun onSecondTick(cfg: ModeConfig) {
        val current = _timerState.value
        val newElapsed = current.totalElapsedSeconds + 1
        var newRemaining = current.cycleRemainingSeconds - 1
        var newCycle = current.cycleNumber
        var spokeCriticalThisSecond = false
        var adreDue = current.adrenalinaDue
        var activeFiveTip = current.currentFiveHsTsTip

        val compRemCfg = cfg.reminders[ReminderKey.TROCAR_COMPRESSOR]
        val compAnchor = lastReminderElapsed[ReminderKey.TROCAR_COMPRESSOR] ?: 0
        val compRemaining = if (compRemCfg != null && compRemCfg.enabled && compRemCfg.intervalSeconds > 20) {
            compRemCfg.intervalSeconds - (newElapsed - compAnchor)
        } else null
        val shouldPreWarnCompressor = cfg.voiceMessages.compressorPreWarningEnabled && (
            (compRemaining == 15 && lastCompressorPreWarnAnchor != compAnchor) ||
                ((compRemCfg == null || !compRemCfg.enabled) && newRemaining == 15 && cfg.cycleDurationSeconds >= 40)
            )

        var cycleJustEnded = false
        if (newRemaining <= 0) {
            cycleJustEnded = true
            audioEngine.playCycleAlarm()
            // Usa flushQueue = false para não interromper avisos de tempo ou 5Hs/5Ts em andamento
            speakProtected(cfg.voiceMessages.cycleCompletedCue, flushQueue = false)
            newCycle += 1
            newRemaining = cfg.cycleDurationSeconds
            thirtySecondAnnouncedCycle = -1
            lastAnnouncementCycleRemaining = -1
            beatStepInPattern = 0
            ventilationWaitUntilMs = 0L
            remainingVentPauseOnPauseMs = 0L
            isVentilationPhase = false
            spokeCriticalThisSecond = true
            lastCriticalSpeechAt = newElapsed
        } else {
            if (newRemaining == 30 && thirtySecondAnnouncedCycle != newCycle) {
                thirtySecondAnnouncedCycle = newCycle
                speakProtected(cfg.voiceMessages.thirtySecondsCue, flushQueue = false)
                spokeCriticalThisSecond = true
                lastCriticalSpeechAt = newElapsed
            } else if (shouldPreWarnCompressor) {
                lastCompressorPreWarnAnchor = compAnchor
                speakProtected(cfg.voiceMessages.compressorPreWarningCue, flushQueue = false)
                spokeCriticalThisSecond = true
                lastCriticalSpeechAt = newElapsed
            } else if (
                cfg.announceTimeEvery10s &&
                newRemaining % 10 == 0 &&
                newRemaining < cfg.cycleDurationSeconds &&
                lastAnnouncementCycleRemaining != newRemaining
            ) {
                lastAnnouncementCycleRemaining = newRemaining
                speakProtected(formatSpokenPtBrTime(newRemaining), flushQueue = false)
                spokeCriticalThisSecond = true
                lastCriticalSpeechAt = newElapsed
            }
        }

        val adreRemCfg = cfg.reminders[ReminderKey.ADRENALINA]
        val adreAnchor = lastReminderElapsed[ReminderKey.ADRENALINA] ?: 0
        val adrePreRemaining = if (adreRemCfg != null && adreRemCfg.enabled && adreRemCfg.intervalSeconds > 35) {
            adreRemCfg.intervalSeconds - (newElapsed - adreAnchor)
        } else null
        if (cfg.voiceMessages.adrenalinaPreWarningEnabled && adrePreRemaining == 30 && lastAdrenalinaPreWarnAnchor != adreAnchor) {
            lastAdrenalinaPreWarnAnchor = adreAnchor
            speakProtected(cfg.voiceMessages.adrenalinaPreWarningCue, flushQueue = false)
            spokeCriticalThisSecond = true
            lastCriticalSpeechAt = newElapsed
        }

        cfg.reminders.forEach { (key, reminder) ->
            if (!reminder.enabled || reminder.intervalSeconds <= 0) return@forEach
            val lastAt = lastReminderElapsed[key] ?: 0
            if (newElapsed - lastAt >= reminder.intervalSeconds) {
                lastReminderElapsed[key] = newElapsed
                if (cycleJustEnded && (key == ReminderKey.VERIFICAR_RITMO || key == ReminderKey.TROCAR_COMPRESSOR)) {
                    return@forEach
                }
                speakProtected(reminder.message, flushQueue = false)
                spokeCriticalThisSecond = true
                lastCriticalSpeechAt = newElapsed
                if (key == ReminderKey.ADRENALINA) {
                    adreDue = true
                }
            }
        }

        val nowMs = SystemClock.elapsedRealtime()
        val isAudioBusyNow = audioEngine.isSpeaking() || nowMs < protectedSpeechUntilMs

        if (
            cfg.pattern.isContinuous &&
            !current.isIntubated &&
            !spokeCriticalThisSecond &&
            !isAudioBusyNow &&
            newElapsed % cfg.continuousVentilationIntervalSeconds == 0
        ) {
            speakProtected(cfg.voiceMessages.ventilationsCue, flushQueue = false)
            spokeCriticalThisSecond = true
            lastCriticalSpeechAt = newElapsed
        }

        // Janela segura para 5Hs e 5Ts: nunca interrompe e nunca colide com avisos de tempo ou eventos
        if (cfg.fiveHsTs.enabled) {
            val cycleElapsed = cfg.cycleDurationSeconds - newRemaining
            val nextTenSecIn = if (cfg.announceTimeEvery10s) {
                if (newRemaining % 10 == 0) 0 else newRemaining % 10
            } else 999
            val nextThirtySecIn = if (newRemaining > 30) newRemaining - 30 else 999
            val nextCompPreWarnIn = if (cfg.voiceMessages.compressorPreWarningEnabled) {
                minOf(
                    if (compRemaining != null && compRemaining > 15) compRemaining - 15 else 999,
                    if ((compRemCfg == null || !compRemCfg.enabled) && newRemaining > 15) newRemaining - 15 else 999
                )
            } else 999
            val nextAdrePreWarnIn = if (cfg.voiceMessages.adrenalinaPreWarningEnabled && adrePreRemaining != null && adrePreRemaining > 30) {
                adrePreRemaining - 30
            } else 999
            var nextReminderIn = 999
            cfg.reminders.forEach { (key, r) ->
                if (r.enabled && r.intervalSeconds > 0) {
                    val dueIn = (r.intervalSeconds - (newElapsed - (lastReminderElapsed[key] ?: 0))).coerceAtLeast(0)
                    nextReminderIn = minOf(nextReminderIn, dueIn)
                }
            }
            val nextScheduledAlertIn = minOf(
                nextTenSecIn,
                nextThirtySecIn,
                newRemaining,
                nextCompPreWarnIn,
                nextAdrePreWarnIn,
                nextReminderIn
            )

            fun canSpeakFiveHs(textToSpeak: String): Boolean {
                if (spokeCriticalThisSecond || isVentilationPhase || isAudioBusyNow) return false
                if (cycleElapsed < 10 || newRemaining <= 35) return false
                if (newElapsed - lastCriticalSpeechAt < 2) return false
                val neededClearSeconds = ((estimateSpeechMs(textToSpeak) + 999L) / 1000L).toInt() + 2
                return nextScheduledAlertIn >= neededClearSeconds
            }

            if (!fiveHsInitialDone && newElapsed >= cfg.fiveHsTs.initialPromptDelaySeconds) {
                if (canSpeakFiveHs(cfg.fiveHsTs.initialPromptMessage)) {
                    fiveHsInitialDone = true
                    lastFiveHsTipAt = newElapsed
                    speakProtected(cfg.fiveHsTs.initialPromptMessage, flushQueue = false)
                }
            } else if (fiveHsInitialDone) {
                if (newElapsed - lastFiveHsTipAt >= cfg.fiveHsTs.tipIntervalSeconds) {
                    pendingFiveHsTip = true
                }
                if (pendingFiveHsTip) {
                    val tips = RcpDefaults.FIVE_HS_AND_FIVE_TS
                    val tip = tips[fiveHsTipIndex % tips.size]
                    if (canSpeakFiveHs(tip.spokenTip)) {
                        fiveHsTipIndex = (fiveHsTipIndex + 1) % tips.size
                        lastFiveHsTipAt = newElapsed
                        pendingFiveHsTip = false
                        activeFiveTip = tip
                        speakProtected(tip.spokenTip, flushQueue = false)
                    }
                }
            }
        }

        val adreCfg = cfg.reminders[ReminderKey.ADRENALINA]
        val adreRem = if (adreCfg != null && adreCfg.enabled) {
            (adreCfg.intervalSeconds - (newElapsed - (lastReminderElapsed[ReminderKey.ADRENALINA] ?: 0)))
                .coerceAtLeast(0)
        } else null

        _timerState.value = current.copy(
            cycleNumber = newCycle,
            cycleRemainingSeconds = newRemaining,
            totalElapsedSeconds = newElapsed,
            adrenalinaRemainingSeconds = adreRem,
            adrenalinaDue = adreDue,
            currentFiveHsTsTip = activeFiveTip
        )
        updateNotification()
    }

    private fun onMetronomeBeat(cfg: ModeConfig) {
        val current = _timerState.value
        if (cfg.pattern.isContinuous || current.isIntubated) {
            ventilationWaitUntilMs = 0L
            remainingVentPauseOnPauseMs = 0L
            isVentilationPhase = false
            audioEngine.playMetronomeBeat(isVentilationPhase = false)
            _timerState.value = current.copy(
                beatTick = current.beatTick + 1,
                currentPhaseLabel = if (current.isIntubated) "Compressões contínuas (Intubado)" else "Compressões contínuas"
            )
            return
        }

        val now = SystemClock.elapsedRealtime()
        if (ventilationWaitUntilMs > 0L) {
            if (now < ventilationWaitUntilMs) {
                val secsLeft = ((ventilationWaitUntilMs - now + 999L) / 1000L).toInt().coerceAtLeast(1)
                isVentilationPhase = true
                // O metrônomo nunca para durante a pausa de ventilação
                audioEngine.playMetronomeBeat(isVentilationPhase = true)
                _timerState.value = current.copy(
                    beatTick = current.beatTick + 1,
                    currentPhaseLabel = "Ventilações (${secsLeft}s)"
                )
                return
            }
            ventilationWaitUntilMs = 0L
            isVentilationPhase = false
            beatStepInPattern = 0
            lastCriticalSpeechAt = current.totalElapsedSeconds
            audioEngine.speak(cfg.voiceMessages.compressionsCue, flushQueue = false)
        }

        val compCount = cfg.pattern.compressionsCount
        if (beatStepInPattern >= compCount) {
            ventilationWaitUntilMs = now + 10_000L
            isVentilationPhase = true
            beatStepInPattern = 0
            lastCriticalSpeechAt = current.totalElapsedSeconds
            audioEngine.playMetronomeBeat(isVentilationPhase = true)
            audioEngine.speak(cfg.voiceMessages.ventilationsCue, flushQueue = false)
            _timerState.value = current.copy(
                beatTick = current.beatTick + 1,
                currentPhaseLabel = "Ventilações (10s)"
            )
            return
        }

        val compIdx = beatStepInPattern
        isVentilationPhase = false
        audioEngine.playMetronomeBeat(isVentilationPhase = false)
        beatStepInPattern = compIdx + 1

        _timerState.value = current.copy(
            beatTick = current.beatTick + 1,
            currentPhaseLabel = "Compressões (${compIdx + 1}/$compCount)"
        )
    }

    private fun promoteToForeground() {
        val notification = buildNotification()
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
            startForeground(
                NOTIFICATION_ID,
                notification,
                ServiceInfo.FOREGROUND_SERVICE_TYPE_MEDIA_PLAYBACK
            )
        } else {
            startForeground(NOTIFICATION_ID, notification)
        }
        isForegroundActive = true
    }

    private fun updateNotification() {
        if (!isForegroundActive) return
        val nm = getSystemService(Context.NOTIFICATION_SERVICE) as NotificationManager
        nm.notify(NOTIFICATION_ID, buildNotification())
    }

    private fun buildNotification(): Notification {
        val state = _timerState.value
        val openAppIntent = PendingIntent.getActivity(
            this,
            0,
            Intent(this, MainActivity::class.java),
            PendingIntent.FLAG_IMMUTABLE or PendingIntent.FLAG_UPDATE_CURRENT
        )

        val toggleActionIntent = PendingIntent.getService(
            this,
            1,
            Intent(this, RcpForegroundService::class.java).apply {
                action = if (state.isPaused) ACTION_RESUME else ACTION_PAUSE
            },
            PendingIntent.FLAG_IMMUTABLE or PendingIntent.FLAG_UPDATE_CURRENT
        )

        val stopActionIntent = PendingIntent.getService(
            this,
            2,
            Intent(this, RcpForegroundService::class.java).apply { action = ACTION_STOP },
            PendingIntent.FLAG_IMMUTABLE or PendingIntent.FLAG_UPDATE_CURRENT
        )

        val statusText = if (state.isPaused) "Pausado" else state.currentPhaseLabel
        return NotificationCompat.Builder(this, CHANNEL_ID)
            .setSmallIcon(android.R.drawable.ic_media_play)
            .setContentTitle("${state.activeMode.label} · Ciclo ${state.cycleNumber}")
            .setContentText("Restante ${formatMmSs(state.cycleRemainingSeconds)} · Total ${formatMmSs(state.totalElapsedSeconds)} · $statusText")
            .setContentIntent(openAppIntent)
            .setOngoing(true)
            .setOnlyAlertOnce(true)
            .addAction(
                if (state.isPaused) android.R.drawable.ic_media_play else android.R.drawable.ic_media_pause,
                if (state.isPaused) "Retomar" else "Pausar",
                toggleActionIntent
            )
            .addAction(android.R.drawable.ic_menu_close_clear_cancel, "Encerrar", stopActionIntent)
            .setStyle(
                MediaStyle()
                    .setMediaSession(mediaSession.sessionToken)
                    .setShowActionsInCompactView(0, 1)
            )
            .build()
    }

    private fun updateMediaPlaybackState(isPlaying: Boolean) {
        val state = if (isPlaying) PlaybackStateCompat.STATE_PLAYING else PlaybackStateCompat.STATE_PAUSED
        mediaSession.setPlaybackState(
            PlaybackStateCompat.Builder()
                .setActions(
                    PlaybackStateCompat.ACTION_PLAY or
                        PlaybackStateCompat.ACTION_PAUSE or
                        PlaybackStateCompat.ACTION_STOP
                )
                .setState(state, _timerState.value.totalElapsedSeconds * 1000L, 1.0f)
                .build()
        )
    }

    private fun acquireWakeLock() {
        if (wakeLock?.isHeld == true) return
        val pm = getSystemService(Context.POWER_SERVICE) as PowerManager
        wakeLock = pm.newWakeLock(PowerManager.PARTIAL_WAKE_LOCK, "CronometroRcp::TimerWakeLock").apply {
            acquire(6 * 60 * 60 * 1000L)
        }
    }

    private fun releaseWakeLock() {
        if (wakeLock?.isHeld == true) {
            wakeLock?.release()
        }
        wakeLock = null
    }

    private fun createNotificationChannel() {
        val manager = getSystemService(Context.NOTIFICATION_SERVICE) as NotificationManager
        val channel = NotificationChannel(
            CHANNEL_ID,
            "Cronômetro RCP em Execução",
            NotificationManager.IMPORTANCE_LOW
        ).apply {
            description = "Mantém cronômetro, metrônomo e alertas de voz ativos com a tela apagada."
        }
        manager.createNotificationChannel(channel)
    }

    override fun onDestroy() {
        timerJob?.cancel()
        metronomeJob?.cancel()
        releaseWakeLock()
        mediaSession.release()
        audioEngine.release()
        serviceScope.cancel()
        super.onDestroy()
    }

    companion object {
        const val CHANNEL_ID = "rcp_media_playback_channel"
        const val NOTIFICATION_ID = 1001
        const val ACTION_PAUSE = "br.com.cronometrorcp.ACTION_PAUSE"
        const val ACTION_RESUME = "br.com.cronometrorcp.ACTION_RESUME"
        const val ACTION_STOP = "br.com.cronometrorcp.ACTION_STOP"
    }
}

    private var tts: TextToSpeech? = TextToSpeech(appContext, this)
    @Volatile
    private var isTtsReady = false
    @Volatile

    @Volatile
    private var currentTargetBpm = 110

    @Volatile
    private var hasAudioFocus = false

    private var mp3Player: MediaPlayer? = null

    private val activeUtteranceIds = ConcurrentHashMap.newKeySet<String>()

    private val speechAudioAttributes = AudioAttributes.Builder()
        .setUsage(AudioAttributes.USAGE_MEDIA)
        .setContentType(AudioAttributes.CONTENT_TYPE_SPEECH)
        .build()

    private val alarmAudioAttributes = AudioAttributes.Builder()
        .setUsage(AudioAttributes.USAGE_MEDIA)

    // Solicita foco como conteúdo de fala usando AUDIOFOCUS_GAIN sem MAY_DUCK,
    // evitando que o sistema Android baixe o volume do MP3, metrônomo ou alarmes.
    private val focusRequest: AudioFocusRequest = AudioFocusRequest.Builder(
        AudioManager.AUDIOFOCUS_GAIN
    )
        .setAudioAttributes(speechAudioAttributes)
        .setWillPauseWhenDucked(false)
        .setAcceptsDelayedFocusGain(false)
        .setOnAudioFocusChangeListener { }
        .build()

    private val compressionClickPcm = synthesizeClick(freqHz = 1320.0, durationMs = 55, gain = 1.0f)
    private val ventilationCuePcm = synthesizeClick(freqHz = 820.0, durationMs = 75, gain = 0.95f)
    private val cycleAlarmPcm = synthesizeCycleAlarm()

    private fun ensureAudioFocus() {
        if (!hasAudioFocus) {
            val res = audioManager.requestAudioFocus(focusRequest)
            hasAudioFocus = res == AudioManager.AUDIOFOCUS_REQUEST_GRANTED
        }
    }

    fun updateAudioSettings(settings: AudioSettings) {
        audioSettings = settings
        if (settings.isMuted) {
            tts?.stop()
        }
        if (settings.mp3UriString != loadedMp3Uri) {
            prepareMp3Player(settings.mp3UriString)
        }
        syncMp3PlayerParams()
    }

    fun setTargetBpm(bpm: Int) {
        currentTargetBpm = bpm
        syncMp3PlayerParams()
    }

    private fun prepareMp3Player(uriString: String?) {
        stopMp3Loop()
        mp3Player?.release()
        mp3Player = null
        loadedMp3Uri = uriString

        if (uriString.isNullOrBlank()) return
        try {
            val player = MediaPlayer().apply {
                setAudioAttributes(musicAudioAttributes)
                setDataSource(appContext, Uri.parse(uriString))
                isLooping = true
                prepare()
            }
            mp3Player = player
            syncMp3PlayerParams()
        } catch (_: Exception) {
            mp3Player = null
        }
    }

    private fun syncMp3PlayerParams(applyPlaybackParams: Boolean = false) {
        val player = mp3Player ?: return
        val settings = audioSettings
        // Volume exclusivo do MP3: nunca é reduzido por voz, alarme ou metrônomo
        val mp3Vol = if (settings.isMuted) 0f else settings.mp3Volume.coerceIn(0f, 1f)
        try {
            player.setVolume(mp3Vol, mp3Vol)
            if (applyPlaybackParams || player.isPlaying) {
                val baseSpeed = getBasePlaybackSpeed(currentTargetBpm)
                val combinedSpeed = (baseSpeed * pllSpeedTrim).coerceIn(0.82f, 1.24f)
                player.playbackParams = player.playbackParams.setSpeed(combinedSpeed)
            }
        } catch (_: Exception) {
        }
    }

    fun getExactMp3Bpm(): Double {
        val exact = if (audioSettings.mp3ExactBpm > 1f) {
            audioSettings.mp3ExactBpm.toDouble()
        } else {
            audioSettings.mp3OriginalBpm.toDouble()
        }
        return exact.coerceIn(85.0, 145.0)
    }

    fun getSyncedEffectiveMp3Bpm(modeBpm: Int): Double = getExactMp3Bpm()

    fun getEffectiveBeatBpm(modeBpm: Int): Double {
        if (wantsMp3Playback()) {
            if (audioSettings.mp3SyncSpeed) return modeBpm.toDouble()
            if (audioSettings.followMusicRhythm) return getExactMp3Bpm().coerceIn(85.0, 145.0)
        }
        return modeBpm.toDouble()
    }

    fun getBasePlaybackSpeed(modeBpm: Int): Float {
        if (!wantsMp3Playback() || !audioSettings.mp3SyncSpeed) return 1.0f
        val effectiveMp3 = getSyncedEffectiveMp3Bpm(modeBpm)
        return (modeBpm.toDouble() / effectiveMp3).toFloat().coerceIn(0.85f, 1.18f)
    }

    fun setPllSpeedTrim(trim: Float) {
        val clamped = trim.coerceIn(0.965f, 1.055f)
        if (kotlin.math.abs(clamped - pllSpeedTrim) < 0.0005f) return
        pllSpeedTrim = clamped
        syncMp3PlayerParams()
    }

    fun wantsMp3Playback(): Boolean {
        val s = audioSettings
        val wants = s.rhythmSource == RhythmAudioSource.MP3_ONLY ||
            s.rhythmSource == RhythmAudioSource.MP3_AND_METRONOME
        return wants && !s.mp3UriString.isNullOrBlank()
    }

    fun prepareMp3ForSyncedStart(targetBpm: Int, seekMs: Int) {
        currentTargetBpm = targetBpm
        pllSpeedTrim = 1.0f
        if (!wantsMp3Playback()) {
            pauseMp3Loop()
            return
        }
        ensureAudioFocus()
        if (mp3Player == null || loadedMp3Uri != audioSettings.mp3UriString) {
            prepareMp3Player(audioSettings.mp3UriString)
        }
        try {
            if (mp3Player?.isPlaying == true) {
                mp3Player?.pause()
            }
            val dur = mp3Player?.duration ?: 0
            val safeSeek = if (dur > 500) seekMs.coerceIn(0, dur - 250) else 0
            mp3Player?.seekTo(safeSeek)
        } catch (_: Exception) {
        }
        syncMp3PlayerParams()
    }

    fun startMp3Now(targetBpm: Int) {

            prepareMp3Player(audioSettings.mp3UriString)

    override fun onInit(status: Int) {
        if (status != TextToSpeech.SUCCESS) {
            isTtsReady = false
            return
        }
        val engine = tts ?: return
        engine.setAudioAttributes(speechAudioAttributes)

        val ptBr = Locale("pt", "BR")
        val res = engine.setLanguage(ptBr)
        if (res == TextToSpeech.LANG_MISSING_DATA || res == TextToSpeech.LANG_NOT_SUPPORTED) {
            val sysRes = engine.setLanguage(Locale.getDefault())
            isTtsReady = sysRes != TextToSpeech.LANG_MISSING_DATA &&
                sysRes != TextToSpeech.LANG_NOT_SUPPORTED
        } else {
            isTtsReady = true
        }

        engine.setSpeechRate(1.06f)
        engine.setOnUtteranceProgressListener(object : UtteranceProgressListener() {
            override fun onStart(utteranceId: String?) {}
            override fun onDone(utteranceId: String?) {
                if (utteranceId != null) activeUtteranceIds.remove(utteranceId)
            }
            @Deprecated("Deprecated in Java")
            override fun onError(utteranceId: String?) {
                if (utteranceId != null) activeUtteranceIds.remove(utteranceId)
            }
            override fun onError(utteranceId: String?, errorCode: Int) {
                if (utteranceId != null) activeUtteranceIds.remove(utteranceId)
            }
        })
    }

    fun isSpeaking(): Boolean = activeUtteranceIds.isNotEmpty() || (tts?.isSpeaking == true)

    fun speak(text: String, flushQueue: Boolean = false) {
        val settings = audioSettings
        if (settings.isMuted || settings.voiceVolume <= 0.01f || text.isBlank()) return

        ensureAudioFocus()

        if (!isTtsReady || tts == null) {
            playPcm(ventilationCuePcm, settings.voiceVolume)
            return
        }

        val params = Bundle().apply {
            putFloat(TextToSpeech.Engine.KEY_PARAM_VOLUME, settings.voiceVolume.coerceIn(0f, 1f))
        }
        val queueMode = if (flushQueue) TextToSpeech.QUEUE_FLUSH else TextToSpeech.QUEUE_ADD
        val utteranceId = UUID.randomUUID().toString()
        if (flushQueue) {
            activeUtteranceIds.clear()
        }
        activeUtteranceIds.add(utteranceId)
        val result = tts?.speak(text, queueMode, params, utteranceId)
        if (result == TextToSpeech.ERROR) {
            activeUtteranceIds.remove(utteranceId)
            playPcm(ventilationCuePcm, settings.voiceVolume)
        }
    }

    fun playMetronomeBeat(isVentilationPhase: Boolean) {
        val settings = audioSettings
        if (settings.isMuted || settings.metronomeVolume <= 0.01f) return
        val pcm = if (isVentilationPhase) ventilationCuePcm else compressionClickPcm
        playPcm(pcm, settings.metronomeVolume.coerceIn(0f, 1.0f))
    }

    fun playCycleAlarm() {
        val settings = audioSettings
        if (settings.isMuted || settings.alarmVolume <= 0.01f) return
        playPcm(cycleAlarmPcm, settings.alarmVolume)
    }

    private fun playPcm(samples: ShortArray, volume: Float) {
        val clamped = volume.coerceIn(0f, 1f)
        scope.launch {
            val minBuf = AudioTrack.getMinBufferSize(
                SAMPLE_RATE,
                AudioFormat.CHANNEL_OUT_MONO,
                AudioFormat.ENCODING_PCM_16BIT
            )
            val byteCount = samples.size * 2
            val track = AudioTrack.Builder()
                .setAudioAttributes(alarmAudioAttributes)
                .setAudioFormat(
                    AudioFormat.Builder()
                        .setEncoding(AudioFormat.ENCODING_PCM_16BIT)
                        .setSampleRate(SAMPLE_RATE)
                        .setChannelMask(AudioFormat.CHANNEL_OUT_MONO)
                        .build()
                )
                .setTransferMode(AudioTrack.MODE_STATIC)
                .setBufferSizeInBytes(maxOf(byteCount, minBuf))
                .build()

            try {
                track.setVolume(clamped)
                track.write(samples, 0, samples.size)
                track.play()
                val durationMs = (samples.size * 1000L) / SAMPLE_RATE
                kotlinx.coroutines.delay(durationMs + 40L)
            } catch (_: Exception) {
            } finally {
                try {
                    track.stop()
                } catch (_: Exception) {
                }
                track.release()
            }
        }
    }

    private fun synthesizeClick(freqHz: Double, durationMs: Int, gain: Float): ShortArray {
        val count = (SAMPLE_RATE * durationMs) / 1000
        return ShortArray(count) { i ->
            val t = i.toDouble() / SAMPLE_RATE
            val env = exp(-t * 24.0)
            val fundamental = sin(2.0 * PI * freqHz * t)
            val harmonic = 0.45 * sin(2.0 * PI * (freqHz * 2.0) * t)
            val raw = (fundamental + harmonic) * env * gain
            val saturated = tanh(raw * 2.1)
            (saturated * Short.MAX_VALUE).toInt().coerceIn(Short.MIN_VALUE.toInt(), Short.MAX_VALUE.toInt()).toShort()
        }
    }

    private fun synthesizeCycleAlarm(): ShortArray {
        val toneMs = 150
        val gapMs = 80
        val totalMs = toneMs * 3 + gapMs * 2
        val count = (SAMPLE_RATE * totalMs) / 1000
        val freqs = listOf(880.0, 1046.5, 1318.5)
        return ShortArray(count) { i ->
            val ms = (i * 1000.0) / SAMPLE_RATE
            val block = (ms / (toneMs + gapMs)).toInt()
            val posInBlock = ms % (toneMs + gapMs)
            if (block < 3 && posInBlock <= toneMs) {
                val t = posInBlock / 1000.0
                val env = (1.0 - (posInBlock / toneMs) * 0.35)
                val wave = sin(2.0 * PI * freqs[block] * t) * env * 0.9
                (wave * Short.MAX_VALUE).toInt().coerceIn(Short.MIN_VALUE.toInt(), Short.MAX_VALUE.toInt()).toShort()
            } else {
                0
            }
        }
    }

    fun release() {
        stopMp3Loop()
        mp3Player?.release()
        mp3Player = null
        tts?.stop()
        tts?.shutdown()
        tts = null
        if (hasAudioFocus) {
            audioManager.abandonAudioFocusRequest(focusRequest)
            hasAudioFocus = false
        }
    }

    companion object {
        private const val SAMPLE_RATE = 22050
    }
}

package br.com.cronometrorcp.audio

import android.content.Context
import android.media.MediaCodec
import android.media.MediaExtractor
import android.media.MediaFormat
import android.net.Uri
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import java.nio.ByteOrder
import kotlin.math.PI
import kotlin.math.ceil
import kotlin.math.floor
import kotlin.math.roundToInt
import kotlin.math.sqrt

data class Mp3BeatAnalysis(
    val detectedBpm: Int = 110,
    val exactBpm: Float = 110f,
    val beatOffsetMs: Int = 0,
    val trimStartMs: Int = 0,
    val loopEndMs: Int = 0
)

object Mp3BeatAnalyzer {

    suspend fun analyzeUri(context: Context, uri: Uri): Mp3BeatAnalysis = withContext(Dispatchers.Default) {
        val extractor = MediaExtractor()
        var codec: MediaCodec? = null
        try {
            extractor.setDataSource(context, uri, null)
            var audioTrackIndex = -1
            var format: MediaFormat? = null
            for (i in 0 until extractor.trackCount) {
                val f = extractor.getTrackFormat(i)
                val mime = f.getString(MediaFormat.KEY_MIME) ?: ""
                if (mime.startsWith("audio/")) {
                    audioTrackIndex = i
                    format = f
                    break
                }
            }
            if (audioTrackIndex < 0 || format == null) return@withContext Mp3BeatAnalysis()

            extractor.selectTrack(audioTrackIndex)
            val mime = format.getString(MediaFormat.KEY_MIME) ?: return@withContext Mp3BeatAnalysis()
            val sampleRate = format.getInteger(MediaFormat.KEY_SAMPLE_RATE)
            val channels = if (format.containsKey(MediaFormat.KEY_CHANNEL_COUNT)) {
                format.getInteger(MediaFormat.KEY_CHANNEL_COUNT)
            } else 1
            val durationUs = if (format.containsKey(MediaFormat.KEY_DURATION)) {
                format.getLong(MediaFormat.KEY_DURATION)
            } else 0L
            val totalDurationSec = (durationUs / 1_000_000.0).coerceAtLeast(0.0)

            // Decodifica até 50 segundos em envelope grave (~155 Hz) e RMS em quadros de ~10 ms
            val hopSamples = (sampleRate / 100).coerceAtLeast(256)
            val maxFrames = 5000
            val bassList = ArrayList<Float>(maxFrames)
            val rmsList = ArrayList<Float>(maxFrames)

            codec = MediaCodec.createDecoderByType(mime).apply {
                configure(format, null, null, 0)
                start()
            }

            var lp = 0.0
            val omega = 2.0 * PI * 155.0
            val lpAlpha = minOf(0.18, omega / (sampleRate + omega))
            var bassAcc = 0.0
            var rmsAcc = 0.0
            var accCount = 0

            val bufferInfo = MediaCodec.BufferInfo()
            var inputDone = false
            var outputDone = false

            while (!outputDone && bassList.size < maxFrames) {
                if (!inputDone) {
                    val inIdx = codec.dequeueInputBuffer(5000)
                    if (inIdx >= 0) {
                        val inBuf = codec.getInputBuffer(inIdx)!!
                        val sampleSize = extractor.readSampleData(inBuf, 0)
                        if (sampleSize < 0) {
                            codec.queueInputBuffer(inIdx, 0, 0, 0L, MediaCodec.BUFFER_FLAG_END_OF_STREAM)
                            inputDone = true
                        } else {
                            codec.queueInputBuffer(inIdx, 0, sampleSize, extractor.sampleTime, 0)
                            extractor.advance()
                        }
                    }
                }

                val outIdx = codec.dequeueOutputBuffer(bufferInfo, 5000)
                if (outIdx >= 0) {
                    val outBuf = codec.getOutputBuffer(outIdx)
                    if (outBuf != null && bufferInfo.size > 0) {
                        outBuf.order(ByteOrder.LITTLE_ENDIAN)
                        val shortBuf = outBuf.asShortBuffer()
                        val totalShorts = bufferInfo.size / 2
                        var i = 0
                        while (i < totalShorts && bassList.size < maxFrames) {
                            val s = shortBuf.get(i) / 32768.0
                            lp += lpAlpha * (s - lp)
                            bassAcc += lp * lp
                            rmsAcc += s * s
                            accCount++
                            if (accCount >= hopSamples) {
                                bassList.add(sqrt(bassAcc / accCount).toFloat())
                                rmsList.add(sqrt(rmsAcc / accCount).toFloat())
                                bassAcc = 0.0
                                rmsAcc = 0.0
                                accCount = 0
                            }
                            i += channels
                        }
                    }
                    codec.releaseOutputBuffer(outIdx, false)
                    if ((bufferInfo.flags and MediaCodec.BUFFER_FLAG_END_OF_STREAM) != 0) {
                        outputDone = true
                    }
                }
            }

            val numFrames = bassList.size
            if (numFrames < 400) return@withContext Mp3BeatAnalysis()

            val nonZero = rmsList.filter { it > 0.002f }.sorted()
            val medianRms = if (nonZero.isNotEmpty()) nonZero[nonZero.size / 2] else 0.05f
            val silenceThresh = maxOf(0.003f, medianRms * 0.08f)

            var firstAudibleFrame = 0
            while (firstAudibleFrame < numFrames && rmsList[firstAudibleFrame] < silenceThresh) {
                firstAudibleFrame++
            }
            val framesPerSec = sampleRate.toDouble() / hopSamples
            val firstAudibleSec = firstAudibleFrame / framesPerSec

            val bpmOnset = FloatArray(numFrames)
            val kickOnset = FloatArray(numFrames)
            for (f in 1 until numFrames) {
                val dBass = (bassList[f] - bassList[f - 1]).coerceAtLeast(0f)
                val dFull = (rmsList[f] - rmsList[f - 1]).coerceAtLeast(0f)
                bpmOnset[f] = dBass * 0.72f + dFull * 0.28f
                kickOnset[f] = dBass * 0.90f + dFull * 0.10f
            }

            fun sampleArr(arr: FloatArray, idxFloat: Double): Double {
                val i0 = floor(idxFloat).toInt()
                val frac = idxFloat - i0
                if (i0 < 0 || i0 + 1 >= numFrames) return 0.0
                return arr[i0] * (1.0 - frac) + arr[i0 + 1] * frac
            }

            fun scoreBpmComb(bpmVal: Double, highLag: Boolean): Double {
                val periodFrames = (60.0 / bpmVal) * framesPerSec
                val maxMult = if (highLag) 24.0 else 8.0
                val limit = numFrames - ceil(periodFrames * maxMult).toInt() - 2
                if (limit <= 20) return -1.0

                var corr = 0.0
                if (highLag) {
                    for (i in 0 until limit) {
                        corr += bpmOnset[i] * (
                            sampleArr(bpmOnset, i + periodFrames) +
                                0.85 * sampleArr(bpmOnset, i + periodFrames * 2.0) +
                                0.7 * sampleArr(bpmOnset, i + periodFrames * 4.0) +
                                0.55 * sampleArr(bpmOnset, i + periodFrames * 8.0) +
                                0.45 * sampleArr(bpmOnset, i + periodFrames * 16.0) +
                                0.35 * sampleArr(bpmOnset, i + periodFrames * 24.0)
                            )
                    }
                } else {
                    for (i in 0 until limit) {
                        corr += bpmOnset[i] * (
                            sampleArr(bpmOnset, i + periodFrames) +
                                0.8 * sampleArr(bpmOnset, i + periodFrames * 2.0) +
                                0.6 * sampleArr(bpmOnset, i + periodFrames * 4.0) +
                                0.45 * sampleArr(bpmOnset, i + periodFrames * 8.0)
                            )
                    }
                }
                return corr / limit
            }

            // Estágio 1: busca focada 92–138 BPM (músicas em torno de 110–120 BPM)
            var coarseBpm = 115.0
            var bestScore = -1.0
            var bpm = 92.0
            while (bpm <= 138.0) {
                val sc = scoreBpmComb(bpm, highLag = false)
                if (sc > bestScore) {
                    bestScore = sc
                    coarseBpm = bpm
                }
                bpm += 0.25
            }

            // Estágio 2: busca fina ±1.5 BPM em passo de 0.005 BPM
            var exactBpm = coarseBpm
            var bestFineScore = -1.0
            val fineLow = maxOf(90.0, coarseBpm - 1.5)
            val fineHigh = minOf(140.0, coarseBpm + 1.5)
            bpm = fineLow
            while (bpm <= fineHigh) {
                val sc = scoreBpmComb(bpm, highLag = true)
                if (sc > bestFineScore) {
                    bestFineScore = sc
                    exactBpm = bpm
                }
                bpm += 0.005
            }

            // Polimento ultrafino ±0.04 BPM em passo de 0.001 BPM
            val polishLow = maxOf(90.0, exactBpm - 0.04)
            val polishHigh = minOf(140.0, exactBpm + 0.04)
            var bestPolishScore = bestFineScore
            bpm = polishLow
            while (bpm <= polishHigh) {
                val sc = scoreBpmComb(bpm, highLag = true)
                if (sc > bestPolishScore) {
                    bestPolishScore = sc
                    exactBpm = bpm
                }
                bpm += 0.001
            }

            exactBpm = (exactBpm * 1000.0).roundToInt() / 1000.0

            // Estágio 3: busca de fase interpolada (1 ms) usando kickOnset
            val periodSec = 60.0 / exactBpm
            val periodMs = periodSec * 1000.0
            var bestOffsetMs = 0
            var bestPhaseScore = -1.0
            var offMs = 0.0
            while (offMs < periodMs) {
                var score = 0.0
                var tMs = offMs
                while (tMs < 44_000.0) {
                    val fFloat = (tMs / 1000.0) * framesPerSec
                    if (fFloat >= 1.0 && fFloat < numFrames - 1) {
                        score += sampleArr(kickOnset, fFloat) * 1.4 +
                            sampleArr(kickOnset, fFloat - 0.8) * 0.35 +
                            sampleArr(kickOnset, fFloat + 0.8) * 0.35
                    }
                    tMs += periodMs
                }
                if (score > bestPhaseScore) {
                    bestPhaseScore = score
                    bestOffsetMs = offMs.roundToInt()
                }
                offMs += 1.0
            }

            fun measureLowEndPunch(candMs: Int): Double {
                var punch = 0.0
                var tMs = candMs.toDouble()
                var count = 0
                while (count < 48 && tMs < 44_000.0) {
                    val idx = (tMs / 1000.0) * framesPerSec
                    if (idx >= 1.0 && idx < numFrames - 1) {
                        punch += sampleArr(kickOnset, idx)
                    }
                    tMs += periodMs
                    count++
                }
                return punch
            }

            val periodInt = periodMs.roundToInt().coerceAtLeast(250)
            val halfCand = (bestOffsetMs + periodInt / 2) % periodInt
            if (measureLowEndPunch(halfCand) > measureLowEndPunch(bestOffsetMs) * 1.15) {
                bestOffsetMs = halfCand
            }

            // Compensação do centro da janela (~6 ms) para coincidir com o início do ataque do bumbo
            val hopLeadMs = (600.0 / framesPerSec).roundToInt()
            bestOffsetMs = (((bestOffsetMs - hopLeadMs) % periodInt) + periodInt) % periodInt

            val offsetSec = bestOffsetMs / 1000.0
            var trimStartMs = 0
            var loopStartSec = offsetSec
            if (firstAudibleSec > 0.25) {
                val kFirst = ceil((firstAudibleSec - offsetSec) / periodSec).toInt()
                loopStartSec = maxOf(offsetSec, offsetSec + kFirst * periodSec)
                trimStartMs = maxOf(0, ((loopStartSec - 0.03) * 1000.0).roundToInt())
            }

            var loopEndMs = 0
            if (totalDurationSec > 8.0) {
                val usableEnd = totalDurationSec - 0.12
                val nBeats = floor((usableEnd - loopStartSec) / periodSec).toInt()
                if (nBeats >= 8) {
                    loopEndMs = ((loopStartSec + nBeats * periodSec) * 1000.0).roundToInt()
                }
            }

            Mp3BeatAnalysis(
                detectedBpm = exactBpm.roundToInt(),
                exactBpm = exactBpm.toFloat(),
                beatOffsetMs = bestOffsetMs,
                trimStartMs = trimStartMs,
                loopEndMs = loopEndMs
            )
        } catch (_: Exception) {
            Mp3BeatAnalysis()
        } finally {
            try {
                codec?.stop()
                codec?.release()
            } catch (_: Exception) {
            }
            try {
                extractor.release()
            } catch (_: Exception) {
            }
        }
    }
}

fun formatMmSs(totalSeconds: Int): String {
    val safe = totalSeconds.coerceAtLeast(0)
    val mins = safe / 60
    val secs = safe % 60
    return "%02d:%02d".format(mins, secs)
}

fun formatSpokenPtBrTime(totalSeconds: Int): String {
    val safe = totalSeconds.coerceAtLeast(0)
    val mins = safe / 60
    val secs = safe % 60

    val minText = when (mins) {
        0 -> ""
        1 -> "Um minuto"
        2 -> "Dois minutos"
        3 -> "Três minutos"
        4 -> "Quatro minutos"
        5 -> "Cinco minutos"
        else -> "$mins minutos"
    }

    val secText = when (secs) {
        0 -> ""
        10 -> "dez segundos"
        20 -> "vinte segundos"
        30 -> "trinta segundos"
        40 -> "quarenta segundos"
        50 -> "cinquenta segundos"
        else -> "$secs segundos"
    }

    return when {
        minText.isNotEmpty() && secText.isNotEmpty() -> "$minText e $secText."
        minText.isNotEmpty() -> "$minText."
        secText.isNotEmpty() -> secText.replaceFirstChar { it.uppercase() } + "."
        else -> "Zero segundos."
    }
}

package br.com.cronometrorcp.viewmodel

import android.app.Application
import android.content.ComponentName
import android.content.Context
import android.content.Intent
import android.content.ServiceConnection
import android.os.IBinder
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import br.com.cronometrorcp.data.RcpRepository
import br.com.cronometrorcp.model.*
import br.com.cronometrorcp.service.RcpForegroundService
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.launch

class RcpViewModel(application: Application) : AndroidViewModel(application) {

    private val repository = RcpRepository(application)

    private val _timerState = MutableStateFlow(RcpTimerState())
    val timerState: StateFlow<RcpTimerState> = _timerState.asStateFlow()

    private val _audioSettings = MutableStateFlow(repository.loadAudioSettings())
    val audioSettings: StateFlow<AudioSettings> = _audioSettings.asStateFlow()

    private val _modeConfigs = MutableStateFlow(repository.loadModeConfigs())
    val modeConfigs: StateFlow<Map<RcpMode, ModeConfig>> = _modeConfigs.asStateFlow()

    private val _history = MutableStateFlow(repository.loadHistory())
    val history: StateFlow<List<RcpSessionRecord>> = _history.asStateFlow()

    private var boundService: RcpForegroundService? = null

    private val connection = object : ServiceConnection {
        override fun onServiceConnected(name: ComponentName?, service: IBinder?) {
            val local = (service as RcpForegroundService.LocalBinder).getService()
            boundService = local
            local.refreshAudioSettings(_audioSettings.value)
            viewModelScope.launch {
                local.timerState.collect { state ->
                    _timerState.value = state
                }
            }
        }

        override fun onServiceDisconnected(name: ComponentName?) {
            boundService = null
        }
    }

    init {
        val app = getApplication<Application>()
        val intent = Intent(app, RcpForegroundService::class.java)
        app.startService(intent)
        app.bindService(intent, connection, Context.BIND_AUTO_CREATE)
    }

    fun selectMode(mode: RcpMode) {
        boundService?.selectMode(mode)
    }

    fun startRcp() {
        boundService?.startRcp()
    }

    fun togglePause() {
        val state = _timerState.value
        if (state.isPaused) boundService?.resumeRcp() else boundService?.pauseRcp()
    }

    fun finishRcp() {
        val saved = boundService?.stopAndSaveSession()
        if (saved != null) {
            _history.value = repository.loadHistory()
        }
    }

    fun registerEvent(type: RcpEventType, customTitle: String? = null) {
        boundService?.registerEvent(type, customTitle)
    }

    fun updateAudioSettings(transform: (AudioSettings) -> AudioSettings) {
        val updated = transform(_audioSettings.value)
        _audioSettings.value = updated
        repository.saveAudioSettings(updated)
        boundService?.refreshAudioSettings(updated)
    }

    fun toggleMute() {
        updateAudioSettings { it.copy(isMuted = !it.isMuted) }
    }

    fun updateModeConfig(mode: RcpMode, config: ModeConfig) {
        val map = _modeConfigs.value.toMutableMap()
        map[mode] = config
        _modeConfigs.value = map
        repository.saveModeConfigs(map)
        if (!_timerState.value.isRunning && _timerState.value.activeMode == mode) {
            _timerState.value = _timerState.value.copy(cycleRemainingSeconds = config.cycleDurationSeconds)
        }
    }

    fun deleteSession(sessionId: String) {
        repository.deleteSession(sessionId)
        _history.value = repository.loadHistory()
    }

    override fun onCleared() {
        super.onCleared()
        try {
            getApplication<Application>().unbindService(connection)
        } catch (_: Exception) {
        }
    }
}

package br.com.cronometrorcp.data

import android.content.Context

class RcpRepository(private val context: Context) {

    private val prefs = context.getSharedPreferences("rcp_prefs_v2", Context.MODE_PRIVATE)
    private val historyFile: File get() = File(context.filesDir, "rcp_history.json")

    fun loadAudioSettings(): AudioSettings {
        val rawSource = prefs.getString("rhythm_source", RhythmAudioSource.METRONOME_ONLY.name)?.let { name ->
            RhythmAudioSource.entries.find { it.name == name }
        } ?: RhythmAudioSource.METRONOME_ONLY
        val resolvedSource = if (rawSource == RhythmAudioSource.MP3_ONLY) {
            RhythmAudioSource.MP3_AND_METRONOME
        } else {
            rawSource
        }
        val origBpm = prefs.getInt("mp3_orig_bpm", 110)
        return AudioSettings(
            voiceVolume = prefs.getFloat("voice_vol_v2", 0.85f),
            metronomeVolume = prefs.getFloat("metro_vol_v2", 1.0f).coerceIn(0f, 1.0f),
            mp3Volume = prefs.getFloat("mp3_vol_v2", 0.55f),
            alarmVolume = prefs.getFloat("alarm_vol_v2", 0.85f),
            isMuted = prefs.getBoolean("is_muted", false),
            rhythmSource = resolvedSource,
            mp3UriString = prefs.getString("mp3_uri", null),
            mp3DisplayName = prefs.getString("mp3_name", null),
            mp3OriginalBpm = origBpm,
            mp3ExactBpm = prefs.getFloat("mp3_exact_bpm", origBpm.toFloat()),
            mp3BeatOffsetMs = prefs.getInt("mp3_beat_offset_ms", 0),
            mp3TrimStartMs = prefs.getInt("mp3_trim_start_ms", 0),
            mp3LoopEndMs = prefs.getInt("mp3_loop_end_ms", 0),
            mp3SyncSpeed = prefs.getBoolean("mp3_sync_speed_v2", true),
            followMusicRhythm = prefs.getBoolean("follow_music_rhythm", true)
        )
    }

    fun saveAudioSettings(settings: AudioSettings) {
        val safeSource = if (settings.rhythmSource == RhythmAudioSource.MP3_ONLY) {
            RhythmAudioSource.MP3_AND_METRONOME
        } else {
            settings.rhythmSource
        }
        prefs.edit()
            .putFloat("voice_vol_v2", settings.voiceVolume)
            .putFloat("metro_vol_v2", settings.metronomeVolume.coerceIn(0f, 1.0f))
            .putFloat("mp3_vol_v2", settings.mp3Volume)
            .putFloat("alarm_vol_v2", settings.alarmVolume)
            .putBoolean("is_muted", settings.isMuted)
            .putString("rhythm_source", safeSource.name)
            .putString("mp3_uri", settings.mp3UriString)
            .putString("mp3_name", settings.mp3DisplayName)
            .putInt("mp3_orig_bpm", settings.mp3OriginalBpm)
            .putFloat("mp3_exact_bpm", settings.mp3ExactBpm)
            .putInt("mp3_beat_offset_ms", settings.mp3BeatOffsetMs)
            .putInt("mp3_trim_start_ms", settings.mp3TrimStartMs)
            .putInt("mp3_loop_end_ms", settings.mp3LoopEndMs)
            .putBoolean("mp3_sync_speed_v2", settings.mp3SyncSpeed)
            .putBoolean("follow_music_rhythm", settings.followMusicRhythm)
            .apply()
    }

    fun loadSelectedMode(): RcpMode {
        val raw = prefs.getString("selected_mode", RcpMode.ADULTO_ACLS.name)
        return RcpMode.entries.find { it.name == raw } ?: RcpMode.ADULTO_ACLS
    }

    fun saveSelectedMode(mode: RcpMode) {

    fun loadModeConfigs(): Map<RcpMode, ModeConfig> {
        val defaults = RcpDefaults.defaultConfigs().toMutableMap()
        val raw = prefs.getString("mode_configs_json_v4", null) ?: return defaults
        return try {
            val root = JSONObject(raw)
            RcpMode.entries.forEach { mode ->
                val obj = root.optJSONObject(mode.name) ?: return@forEach
                val base = defaults[mode]!!
                val msgObj = obj.optJSONObject("voiceMessages")
                val remObj = obj.optJSONObject("reminders")
                val amioObj = obj.optJSONObject("amiodaronaProtocol")
                val fiveObj = obj.optJSONObject("fiveHsTs")

                val reminders = base.reminders.toMutableMap()
                ReminderKey.entries.forEach { key ->
                    remObj?.optJSONObject(key.name)?.let { r ->
                        reminders[key] = VoiceReminderConfig(
                            enabled = r.optBoolean("enabled", reminders[key]!!.enabled),
                            intervalSeconds = r.optInt("intervalSeconds", reminders[key]!!.intervalSeconds),
                            message = r.optString("message", reminders[key]!!.message)
                        )
                    }
                }
                defaults[mode] = base.copy(
                    cycleDurationSeconds = obj.optInt("cycleDurationSeconds", base.cycleDurationSeconds),
                    metronomeBpm = obj.optInt("metronomeBpm", base.metronomeBpm),
                    pattern = CompressionPattern.entries.find {
                        it.name == obj.optString("pattern", base.pattern.name)
                    } ?: base.pattern,
                    rhythmCheckIntervalSeconds = obj.optInt("rhythmCheckIntervalSeconds", base.rhythmCheckIntervalSeconds),
                    continuousVentilationIntervalSeconds = obj.optInt(
                        "continuousVentilationIntervalSeconds",
                        base.continuousVentilationIntervalSeconds
                    ),
                    announceTimeEvery10s = obj.optBoolean("announceTimeEvery10s", base.announceTimeEvery10s),
                    voiceMessages = if (msgObj != null) {
                        val savedPreCue = msgObj.optString("compressorPreWarningCue", base.voiceMessages.compressorPreWarningCue)
                        VoiceMessages(
                            compressionsCue = msgObj.optString("compressionsCue", base.voiceMessages.compressionsCue),
                            ventilationsCue = msgObj.optString("ventilationsCue", base.voiceMessages.ventilationsCue),
                            thirtySecondsCue = msgObj.optString("thirtySecondsCue", base.voiceMessages.thirtySecondsCue),
                            compressorPreWarningEnabled = msgObj.optBoolean("compressorPreWarningEnabled", base.voiceMessages.compressorPreWarningEnabled),
                            compressorPreWarningCue = if (savedPreCue == "Dez segundos para trocar o compressor.") {
                                base.voiceMessages.compressorPreWarningCue
                            } else savedPreCue,
                            adrenalinaPreWarningEnabled = msgObj.optBoolean("adrenalinaPreWarningEnabled", base.voiceMessages.adrenalinaPreWarningEnabled),
                            adrenalinaPreWarningCue = msgObj.optString("adrenalinaPreWarningCue", base.voiceMessages.adrenalinaPreWarningCue),
                            cycleCompletedCue = msgObj.optString("cycleCompletedCue", base.voiceMessages.cycleCompletedCue)
                        )
                    } else base.voiceMessages,
                    reminders = reminders,
                    amiodaronaProtocol = if (amioObj != null) AmiodaronaProtocolConfig(
                        enabled = amioObj.optBoolean("enabled", base.amiodaronaProtocol.enabled),
                        firstDoseAfterShock = amioObj.optInt("firstDoseAfterShock", base.amiodaronaProtocol.firstDoseAfterShock),
                        secondDoseAfterShock = amioObj.optInt("secondDoseAfterShock", base.amiodaronaProtocol.secondDoseAfterShock),
                        firstDoseMessage = amioObj.optString("firstDoseMessage", base.amiodaronaProtocol.firstDoseMessage),
                        secondDoseMessage = amioObj.optString("secondDoseMessage", base.amiodaronaProtocol.secondDoseMessage)
                    ) else base.amiodaronaProtocol,
                    fiveHsTs = if (fiveObj != null) {
                        val savedFiveInit = fiveObj.optString("initialPromptMessage", base.fiveHsTs.initialPromptMessage)
                        FiveHsTsConfig(
                            enabled = fiveObj.optBoolean("enabled", base.fiveHsTs.enabled),
                            initialPromptDelaySeconds = fiveObj.optInt("initialPromptDelaySeconds", base.fiveHsTs.initialPromptDelaySeconds),
                            tipIntervalSeconds = fiveObj.optInt("tipIntervalSeconds", base.fiveHsTs.tipIntervalSeconds),
                            initialPromptMessage = if (savedFiveInit == "Lembrete de causas reversíveis: avalie os cinco agás e os cinco tês durante a RCP.") {
                                base.fiveHsTs.initialPromptMessage
                            } else savedFiveInit
                        )
                    } else base.fiveHsTs
                )
            }
            defaults
        } catch (_: Exception) {
            defaults
        }
    }

    fun saveModeConfigs(configs: Map<RcpMode, ModeConfig>) {
        val root = JSONObject()
        configs.forEach { (mode, cfg) ->
            val msgObj = JSONObject().apply {

                put("cycleCompletedCue", cfg.voiceMessages.cycleCompletedCue)
            }
            val remObj = JSONObject()
            cfg.reminders.forEach { (key, r) ->
                remObj.put(key.name, JSONObject().apply {
                    put("enabled", r.enabled)
                    put("intervalSeconds", r.intervalSeconds)
                    put("message", r.message)
                })
            }
            val amioObj = JSONObject().apply {
                put("enabled", cfg.amiodaronaProtocol.enabled)
                put("firstDoseAfterShock", cfg.amiodaronaProtocol.firstDoseAfterShock)
                put("secondDoseAfterShock", cfg.amiodaronaProtocol.secondDoseAfterShock)
                put("firstDoseMessage", cfg.amiodaronaProtocol.firstDoseMessage)
                put("secondDoseMessage", cfg.amiodaronaProtocol.secondDoseMessage)
            }
            val fiveObj = JSONObject().apply {
                put("enabled", cfg.fiveHsTs.enabled)
                put("initialPromptDelaySeconds", cfg.fiveHsTs.initialPromptDelaySeconds)
                put("tipIntervalSeconds", cfg.fiveHsTs.tipIntervalSeconds)
                put("initialPromptMessage", cfg.fiveHsTs.initialPromptMessage)
            }
            root.put(mode.name, JSONObject().apply {
                put("cycleDurationSeconds", cfg.cycleDurationSeconds)
                put("metronomeBpm", cfg.metronomeBpm)
                put("pattern", cfg.pattern.name)
                put("rhythmCheckIntervalSeconds", cfg.rhythmCheckIntervalSeconds)
                put("continuousVentilationIntervalSeconds", cfg.continuousVentilationIntervalSeconds)
                put("announceTimeEvery10s", cfg.announceTimeEvery10s)
                put("voiceMessages", msgObj)
                put("reminders", remObj)
                put("amiodaronaProtocol", amioObj)
                put("fiveHsTs", fiveObj)
            })
        }
        prefs.edit().putString("mode_configs_json_v4", root.toString()).apply()
    }

    fun loadHistory(): List<RcpSessionRecord> {
        if (!historyFile.exists()) return emptyList()
        return try {
            val arr = JSONArray(historyFile.readText())
            (0 until arr.length()).map { idx ->
                val o = arr.getJSONObject(idx)
                val eventsArr = o.optJSONArray("events") ?: JSONArray()
                val events = (0 until eventsArr.length()).map { eIdx ->
                    val eo = eventsArr.getJSONObject(eIdx)
                    val rawType = eo.getString("type")
                    val resolvedType = if (rawType == "EPINEFRINA") {
                        RcpEventType.ADRENALINA
                    } else {
                        RcpEventType.entries.find { it.name == rawType } ?: RcpEventType.PERSONALIZADO
                    }
                    RcpEventRecord(
                        id = eo.getString("id"),
                        type = resolvedType,
                        customTitle = eo.optString("customTitle").takeIf { it.isNotBlank() }
                            ?: if (resolvedType != RcpEventType.PERSONALIZADO) null else rawType,
                        elapsedSeconds = eo.getInt("elapsedSeconds"),
                        cycleNumber = eo.getInt("cycleNumber"),
                        timestampIso = eo.getString("timestampIso")
                    )
                }
                RcpSessionRecord(
                    id = o.getString("id"),
                    startedAtIso = o.getString("startedAtIso"),
                    mode = RcpMode.valueOf(o.getString("mode")),
                    totalDurationSeconds = o.getInt("totalDurationSeconds"),
                    completedCycles = o.getInt("completedCycles"),
                    shocksCount = o.getInt("shocksCount"),
                    medicationsCount = o.getInt("medicationsCount"),
                    compressorSwitchesCount = o.getInt("compressorSwitchesCount"),
                    events = events
                )
            }
        } catch (_: Exception) {
            emptyList()
        }
    }

    fun appendSession(record: RcpSessionRecord) {
        val current = loadHistory().toMutableList()
        current.add(0, record)
        saveHistoryList(current)
    }

    fun deleteSession(sessionId: String) {
        val filtered = loadHistory().filterNot { it.id == sessionId }
        saveHistoryList(filtered)
    }

    private fun saveHistoryList(list: List<RcpSessionRecord>) {
        val arr = JSONArray()
        list.forEach { s ->
            val eventsArr = JSONArray()
            s.events.forEach { e ->
                eventsArr.put(JSONObject().apply {
                    put("id", e.id)
                    put("type", e.type.name)
                    put("customTitle", e.customTitle ?: "")
                    put("elapsedSeconds", e.elapsedSeconds)
                    put("cycleNumber", e.cycleNumber)
                    put("timestampIso", e.timestampIso)
                })
            }
            arr.put(JSONObject().apply {
                put("id", s.id)
                put("startedAtIso", s.startedAtIso)
                put("mode", s.mode.name)
                put("totalDurationSeconds", s.totalDurationSeconds)
                put("completedCycles", s.completedCycles)
                put("shocksCount", s.shocksCount)
                put("medicationsCount", s.medicationsCount)
                put("compressorSwitchesCount", s.compressorSwitchesCount)
                put("events", eventsArr)
            })
        }
        historyFile.writeText(arr.toString())
    }
}

    private val mp3PickerLauncher = registerForActivityResult(

    ) { uri: Uri? ->
        if (uri != null) {
            try {
                contentResolver.takePersistableUriPermission(
                    uri,
                    Intent.FLAG_GRANT_READ_URI_PERMISSION
                )
            } catch (_: Exception) {
            }
            val displayName = resolveDisplayName(uri)
            val newUriStr = uri.toString()
            viewModel.updateAudioSettings { current ->
                current.copy(
                    mp3UriString = newUriStr,
                    mp3DisplayName = displayName,
                    mp3BeatOffsetMs = 0,
                    mp3TrimStartMs = 0,
                    mp3LoopEndMs = 0,
                    mp3SyncSpeed = true,
                    followMusicRhythm = true,
                    rhythmSource = if (current.rhythmSource == RhythmAudioSource.METRONOME_ONLY) {
                        RhythmAudioSource.MP3_AND_METRONOME
                    } else {
                        current.rhythmSource
                    }
                )
            }
            lifecycleScope.launch {
                val analysis = Mp3BeatAnalyzer.analyzeUri(this@MainActivity, uri)
                viewModel.updateAudioSettings { current ->
                    if (current.mp3UriString != newUriStr) return@updateAudioSettings current
                    current.copy(
                        mp3OriginalBpm = analysis.detectedBpm,
                        mp3ExactBpm = analysis.exactBpm,
                        mp3BeatOffsetMs = analysis.beatOffsetMs,
                        mp3TrimStartMs = analysis.trimStartMs,
                        mp3LoopEndMs = analysis.loopEndMs,
                        mp3SyncSpeed = true,
                        followMusicRhythm = true
                    )
                }
            }
        }
    }

    private fun resolveDisplayName(uri: Uri): String {
        contentResolver.query(uri, arrayOf(OpenableColumns.DISPLAY_NAME), null, null, null)?.use { cursor ->
            if (cursor.moveToFirst()) {
                val idx = cursor.getColumnIndex(OpenableColumns.DISPLAY_NAME)
                if (idx >= 0) return cursor.getString(idx) ?: "faixa_ritmo.mp3"
            }
        }
        return uri.lastPathSegment ?: "faixa_ritmo.mp3"
    }

    override fun onCreate(savedInstanceState: Bundle?) {

        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.TIRAMISU) {

                this,
                Manifest.permission.POST_NOTIFICATIONS

                val timerState by viewModel.timerState.collectAsState()
                val audioSettings by viewModel.audioSettings.collectAsState()
                val modeConfigs by viewModel.modeConfigs.collectAsState()
                val history by viewModel.history.collectAsState()

                var route by remember { mutableStateOf(AppRoute.MAIN) }

                LaunchedEffect(timerState.isRunning) {
                    if (timerState.isRunning) {
                        window.addFlags(WindowManager.LayoutParams.FLAG_KEEP_SCREEN_ON)
                        route = AppRoute.MAIN
                    } else {
                        window.clearFlags(WindowManager.LayoutParams.FLAG_KEEP_SCREEN_ON)
                    }
                }

                when (route) {
                    AppRoute.MAIN -> MainScreen(
                        timerState = timerState,
                        audioSettings = audioSettings,
                        activeConfig = modeConfigs[timerState.activeMode]!!,
                        onSelectMode = viewModel::selectMode,
                        onStartRcp = viewModel::startRcp,
                        onTogglePause = viewModel::togglePause,
                        onFinishRcp = viewModel::finishRcp,
                        onToggleMute = viewModel::toggleMute,
                        onUpdateAudio = viewModel::updateAudioSettings,
                        onPickMp3File = { mp3PickerLauncher.launch(arrayOf("audio/*")) },
                        onRegisterEvent = viewModel::registerEvent,
                        onOpenSettings = { if (!timerState.isRunning) route = AppRoute.SETTINGS },
                        onOpenHistory = { if (!timerState.isRunning) route = AppRoute.HISTORY }
                    )

                    AppRoute.SETTINGS -> SettingsScreen(
                        modeConfigs = modeConfigs,
                        audioSettings = audioSettings,
                        onUpdateModeConfig = viewModel::updateModeConfig,

                                it.copy(
                                    mp3UriString = null,
                                    mp3DisplayName = null,
                                    rhythmSource = RhythmAudioSource.METRONOME_ONLY
                                )
                            }
                        },
                        onBack = { route = AppRoute.MAIN }
                    )

                    AppRoute.HISTORY -> HistoryScreen(
                        history = history,
                        onDeleteSession = viewModel::deleteSession,
                        onBack = { route = AppRoute.MAIN }
                    )
                }
            }
        }
    }

# Cronômetro RCP com Áudio — Projeto Nativo Android (Kotlin + Jetpack Compose)

Ferramenta de apoio ao treinamento e à organização de RCP. Funciona **100% offline**, sem acesso à internet e sem APIs externas. Não toma decisões clínicas nem substitui protocolos institucionais.

## Arquitetura Implementada

- **Linguagem & UI**: Kotlin + Jetpack Compose (Material 3 Dark, alto contraste para uso em sala escura ou treinamento).
- **Serviço em Primeiro Plano (`mediaPlayback`)**: `RcpForegroundService` mantém o cronômetro (baseado em `SystemClock.elapsedRealtime()` para zero deriva), o metrônomo PCM e os alertas de voz funcionando mesmo com a tela apagada, com notificação persistente e `PARTIAL_WAKE_LOCK`.
- **Tela Mantida Acesa**: `FLAG_KEEP_SCREEN_ON` gerenciado dinamicamente na `MainActivity` durante uso ativo.
- **Áudio e Voz Offline (`RcpAudioEngine`)**:
  - `TextToSpeech` configurado para `Locale("pt", "BR")` com fallback automático para a voz padrão do sistema caso `pt-BR` esteja indisponível, além de fallback acústico por `AudioTrack`.
  - Solicitação de foco de áudio (`AudioFocusRequest`) com `AudioAttributes.CONTENT_TYPE_SPEECH`.
  - Metrônomo e alarmes gerados por síntese PCM (`AudioTrack`) 100% offline, com volumes independentes para **Voz**, **Metrônomo** e **Alarmes**, além do botão **Silenciar áudio**.
- **Modos Independentes**: Adulto — ACLS, Pediátrico — PALS e Neonatal, cada um com configuração própria de ciclo, BPM, relação compressão/ventilação, intervalos de checagem, lembretes de medicação e mensagens de voz.
- **Persistência Local**: Histórico completo de atendimentos salvo localmente em JSON privado do app (`context.filesDir`), sem rede.

## Como Compilar no Android Studio

1. Crie um novo projeto Empty Activity (Jetpack Compose) no Android Studio (API mínima 26, Target/Compile SDK 35) com pacote `br.com.cronometrorcp`.
2. Copie os arquivos desta pasta para `app/` preservando a estrutura de pacotes.
3. Sincronize o Gradle e execute em um dispositivo físico ou emulador Android.
